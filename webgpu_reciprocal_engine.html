Finds the nearest token ID out of 151,936 instantly without looping.

@group(0) @binding(0) var vocab: array>;
@group(0) @binding(1) var min_dist_cache: array>; // [dist_sq, token_id]
@group(0) @binding(2) var target_pos: vec4;

@compute @workgroup_size(256)
fn reduce_argmin(@builtin(global_invocation_id) global_id: vec3, 
                 @builtin(local_invocation_id) local_id: vec3) {
    let idx = global_id.x;
    
    // 1. Calculate distance for this specific thread's token
    var dist_sq = 999999.0;
    var token_id = f32(idx);
    
    if (idx < 151936u) {
        let diff = vocab[idx].xyz - target_pos.xyz;
        dist_sq = dot(diff, diff);
    }
    
    // 2. Perform local Workgroup reduction (Fold in half iteratively)
    // (Pseudocode for workgroupBarrier() reduction into shared memory)
    // shared_mem[local_id.x] = vec2(dist_sq, token_id);
    // workgroupBarrier();
    // ... fold 256 -> 128 -> 64 -> 32 -> 1 ...
    
    // 3. Write winner of this block to global cache
    if (local_id.x == 0u) {
        min_dist_cache[workgroup_id.x] = shared_mem[0];
    }
}
